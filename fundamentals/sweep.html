INT8 does not automatically reduce latency - It often gets worse until conditions are right.¶

Q/DQ layers in ONNX are for behavior study only.
They do NOT control hardware execution.

Actual execution is decided by the runtime:
ONNX Runtime, TensorRT, or TensorRT-LLM.

Key facts:
- Q/DQ does NOT make weights INT8; they remain FP32.
- ONNX/ORT (Microsoft) and TensorRT/TRT-LLM (NVIDIA) are separate systems.
  Do NOT assume seamless behavior.
- Runtime metrics are the source of truth, not graph annotations.

For INT8 to win:
- Use TensorRT or TensorRT-LLM to compile a real engine.
- Compute must dominate (large layers, large batch): my new MLP is compiled to 0.9GB and start outperform FP16 after BS>=128.
- INT8 has overhead; small models or streaming/low-batch inference
  often run faster in FP16.

import torch
import torch.nn as nn
import torch.nn.functional as F
from typing import Dict, List, Callable
from collections import defaultdict


# ============== QUANTIZATION CORE ==============

class TensorQuantizer(nn.Module):
    """Minimal quantizer using PyTorch fake_quant"""
    
    def __init__(self, num_bits: int = 8, axis: int = None):
        super().__init__()
        self.num_bits = num_bits
        self.axis = axis
        self.register_buffer("amax", None)
    
    def set_amax(self, amax: torch.Tensor):
        self.amax = amax.detach()
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if self.amax is None:
            return x
        scale = (self.amax / 127.0).clamp(min=1e-8)
        if self.axis is None:
            return torch.fake_quantize_per_tensor_affine(
                x, scale.item(), 0, -128, 127
            )
        else:
            return torch.fake_quantize_per_channel_affine(
                x, scale.squeeze(), torch.zeros_like(scale.squeeze(), dtype=torch.int32),
                self.axis, -128, 127
            )


class AmaxListener:
    """Collects max absolute value"""
    
    def __init__(self, axis: int = None):
        self.axis = axis
        self.amax = None
    
    def __call__(self, x: torch.Tensor):
        with torch.no_grad():
            if self.axis is None:
                amax = x.abs().amax()
            else:
                dims = [i for i in range(x.dim()) if i != self.axis]
                amax = x.abs().amax(dim=dims, keepdim=True)
            
            if self.amax is None:
                self.amax = amax
            else:
                self.amax = torch.max(self.amax, amax)


# ============== WRAPPER BASE ==============

class ModuleWrapperMixin:
    """Base mixin for wrappers"""
    
    MODE_PASS_THROUGH = "pass_through"
    MODE_LISTEN = "listen"
    MODE_ACT = "act"
    
    def init_wrapper(self):
        self._mode = self.MODE_PASS_THROUGH
        self._listeners: Dict[str, List[Callable]] = defaultdict(list)
    
    def set_mode(self, mode: str):
        self._mode = mode
    
    def add_listener(self, name: str, listener: Callable):
        self._listeners[name].append(listener)
    
    def get_listeners(self, name: str) -> List[Callable]:
        return self._listeners[name]
    
    def finalize_listening(self):
        raise NotImplementedError


# ============== SMOOTHQUANT WRAPPER ==============

class SmoothQuantLinear(ModuleWrapperMixin, nn.Module):
    """Minimal SmoothQuant wrapper"""
    
    def __init__(self, linear: nn.Linear, alpha: float = 0.5):
        nn.Module.__init__(self)
        self.init_wrapper()
        
        # Store original
        self.linear = linear
        self.alpha = alpha
        
        # Quantizers
        self.input_quantizer = TensorQuantizer(axis=None)      # per-tensor
        self.weight_quantizer = TensorQuantizer(axis=0)        # per-row
        
        # Smooth scale
        self.register_buffer("input_scale", torch.ones(linear.in_features))
        
        # Listeners
        self.add_listener("input", AmaxListener(axis=-1))      # per-channel
        self.add_listener("weight", AmaxListener(axis=1))      # per input channel
    
    @classmethod
    def wrap(cls, module: nn.Linear, **kwargs) -> "SmoothQuantLinear":
        assert isinstance(module, nn.Linear)
        return cls(module, **kwargs)
    
    def finalize_listening(self):
        with torch.no_grad():
            act_listener = self.get_listeners("input")[0]
            weight_listener = self.get_listeners("weight")[0]
            
            act_scale = act_listener.amax.squeeze()
            weight_scale = weight_listener.amax.squeeze()
            
            # SmoothQuant formula
            smooth = (act_scale.pow(self.alpha) / 
                     weight_scale.pow(1 - self.alpha)).clamp(0.01, 100)
            
            # Apply smoothing
            self.linear.weight.data *= smooth.unsqueeze(0)
            self.input_scale = 1.0 / smooth
            
            # Recompute amax after smoothing
            self.input_quantizer.set_amax((act_scale / smooth).amax())
            self.weight_quantizer.set_amax(
                self.linear.weight.abs().amax(dim=1, keepdim=True)
            )
            
            self.linear.weight.data = self.weight_quantizer(self.linear.weight.data)
            
            self.set_mode(self.MODE_ACT)
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if self._mode == self.MODE_PASS_THROUGH:
            return F.linear(x, self.linear.weight, self.linear.bias)
        
        elif self._mode == self.MODE_LISTEN:
            for listener in self.get_listeners("input"):
                listener(x)
            for listener in self.get_listeners("weight"):
                listener(self.linear.weight)
            return F.linear(x, self.linear.weight, self.linear.bias)
        
        elif self._mode == self.MODE_ACT:
            x = x * self.input_scale
            x = self.input_quantizer(x)            
            return F.linear(x, self.linear.weight, self.linear.bias)
        
        else:
            raise ValueError(f"Unknown mode: {self._mode}")


# ============== UTILITIES ==============

def wrap_model(model: nn.Module, wrapper_cls=SmoothQuantLinear, **kwargs):
    """Recursively wrap all nn.Linear"""
    for name, module in model.named_children():
        if isinstance(module, nn.Linear):
            setattr(model, name, wrapper_cls.wrap(module, **kwargs))
        else:
            wrap_model(module, wrapper_cls, **kwargs)
    return model


def set_mode(model: nn.Module, mode: str):
    """Set mode for all wrappers"""
    for module in model.modules():
        if isinstance(module, ModuleWrapperMixin):
            module.set_mode(mode)


def finalize(model: nn.Module):
    """Finalize all wrappers"""
    for module in model.modules():
        if isinstance(module, ModuleWrapperMixin):
            module.finalize_listening()


# ============== MLP MODEL ==============

class MLP(nn.Module):
    def __init__(self, d_model: int, d_hidden: int, n_classes: int):
        super().__init__()
        self.fc1 = nn.Linear(d_model, d_hidden)
        self.fc2 = nn.Linear(d_hidden, d_hidden)
        self.fc3 = nn.Linear(d_hidden, n_classes)
    
    def forward(self, x):
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        return self.fc3(x)

TRT_LOGGER = trt.Logger(trt.Logger.INFO)

def build_engine(onnx_path, engine_path, trt_format="fp16"):
    builder = trt.Builder(TRT_LOGGER)

    network = builder.create_network(
        1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
    )

    parser = trt.OnnxParser(network, TRT_LOGGER)
    with open(onnx_path, "rb") as f:
        assert parser.parse(f.read())

    config = builder.create_builder_config()
    if trt_format=="int8":
        config.set_flag(trt.BuilderFlag.INT8)
    elif trt_format=="fp16":
        config.set_flag(trt.BuilderFlag.FP16)  # allow fallback
    else: print(f"Unsupported format type {trt_format}! Go w. either int8 or fp16")

    # REQUIRED for dynamic batch
    profile = builder.create_optimization_profile()
    profile.set_shape("input", (1, 2560), (16, 2560), (1280, 2560))
    config.add_optimization_profile(profile)

    engine_bytes = builder.build_serialized_network(network, config)
    assert engine_bytes is not None, "Engine build failed"

    with open(engine_path, "wb") as f:
        f.write(engine_bytes)

    print(f"✓ Saved INT8 engine: {engine_path}")

import tensorrt as trt
import numpy as np
from cuda import cudart
import time

TRT_LOGGER = trt.Logger(trt.Logger.INFO)

# -------------------- Benchmarks --------------------

def bench_kernel_only(ctx, bindings, iters=500):
    for _ in range(50):  # warmup
        ctx.execute_v2(bindings)
    cudart.cudaDeviceSynchronize()

    t0 = time.time()
    for _ in range(iters):
        ctx.execute_v2(bindings)
    cudart.cudaDeviceSynchronize()

    return (time.time() - t0) * 1e6 / iters  # µs


def bench_e2e(ctx, bindings, iters=200):
    t0 = time.time()
    for _ in range(iters):
        ctx.execute_v2(bindings)
    cudart.cudaDeviceSynchronize()
    return (time.time() - t0) * 1e6 / iters  # µs


def bench_throughput(ctx, bindings, batch, iters=500):
    t0 = time.time()
    for _ in range(iters):
        ctx.execute_v2(bindings)
    cudart.cudaDeviceSynchronize()
    return batch * iters / (time.time() - t0)  # samples/sec


def effective_bandwidth(bytes_moved, kernel_us):
    return bytes_moved / (kernel_us * 1e-6) / 1e9  # GB/s


# -------------------- Estimation --------------------

def estimate_bytes(engine, batch):
    """
    Crude but consistent estimate:
    input + output + weights
    """
    io_bytes = 0

    for name in engine:
        if engine.get_tensor_mode(name) in (
            trt.TensorIOMode.INPUT,
            trt.TensorIOMode.OUTPUT,
        ):
            dtype = trt.nptype(engine.get_tensor_dtype(name))
            shape = engine.get_tensor_shape(name)
            io_bytes += batch * shape[1] * np.dtype(dtype).itemsize

    weight_bytes = engine.device_memory_size
    return io_bytes + weight_bytes


# -------------------- Sweep --------------------

def sweep(engine_path, batches):
    with open(engine_path, "rb") as f:
        engine = trt.Runtime(TRT_LOGGER).deserialize_cuda_engine(f.read())

    ctx = engine.create_execution_context()

    print(f"\n=== {engine_path} ===")

    for B in batches:
        ctx.set_input_shape("input", (B, 2560))

        bindings = []
        dev = {}

        for name in engine:
            shape = ctx.get_tensor_shape(name)
            if any(d < 0 for d in shape):
                raise RuntimeError(f"Unresolved shape for {name}: {shape}")

            dtype = trt.nptype(engine.get_tensor_dtype(name))
            size = int(np.prod(shape) * np.dtype(dtype).itemsize)

            _, dptr = cudart.cudaMalloc(size)
            dev[name] = dptr
            bindings.append(int(dptr))

        # ---- Measurements ----
        kernel_us = bench_kernel_only(ctx, bindings)
        e2e_us = bench_e2e(ctx, bindings)
        tput = bench_throughput(ctx, bindings, B)

        bytes_moved = estimate_bytes(engine, B)
        bw = effective_bandwidth(bytes_moved, kernel_us)

        print(
            f"B={B:<4d} | "
            f"kernel={kernel_us:8.2f} µs | "
            f"e2e={e2e_us:8.2f} µs | "
            f"tput={tput/1e6:7.2f} M/s | "
            f"bw={bw:7.1f} GB/s"
        )

        for d in dev.values():
            cudart.cudaFree(d)


# -------------------- Run --------------------

batches = [1, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 1280]

sweep("mlp_fp16.trt", batches)
sweep("mlp_trt_int8.trt", batches)

# no q/dq just trt int8

import tensorrt as trt
import numpy as np
from cuda import cudart

class Int8Calibrator(trt.IInt8EntropyCalibrator2):
    def __init__(self, data, input_name="input", cache="calib.cache"):
        super().__init__()
        self.data = data          # shape: [N, B, D]
        self.input_name = input_name
        self.cache = cache
        self.idx = 0

        _, self.d_input = cudart.cudaMalloc(data[0].nbytes)

    def get_batch_size(self):
        return self.data.shape[1]

    def get_batch(self, names):
        if self.idx >= len(self.data):
            return None
        batch = self.data[self.idx]
        cudart.cudaMemcpy(
            self.d_input,
            batch.ctypes.data,
            batch.nbytes,
            cudart.cudaMemcpyKind.cudaMemcpyHostToDevice,
        )
        self.idx += 1
        return [int(self.d_input)]

    def read_calibration_cache(self):
        try:
            return open(self.cache, "rb").read()
        except:
            return None

    def write_calibration_cache(self, cache):
        open(self.cache, "wb").write(cache)

def build_trt_int8(onnx_path, engine_path, calib_data):
    logger = trt.Logger(trt.Logger.INFO)
    builder = trt.Builder(logger)

    network = builder.create_network(
        1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
    )
    parser = trt.OnnxParser(network, logger)

    with open(onnx_path, "rb") as f:
        assert parser.parse(f.read())

    config = builder.create_builder_config()
    config.set_flag(trt.BuilderFlag.INT8)
    config.set_flag(trt.BuilderFlag.FP16)   # fallback OK
    config.int8_calibrator = Int8Calibrator(calib_data)

    profile = builder.create_optimization_profile()
    profile.set_shape("input", (1,2560), (64,2560), (1280,2560))
    config.add_optimization_profile(profile)

    engine = builder.build_serialized_network(network, config)
    open(engine_path, "wb").write(engine)

# shape: [num_batches, batch, 2560]
calib_data = np.random.randn(100, 64, 2560).astype(np.float32)

build_trt_int8(
    onnx_path="mlp.onnx",
    engine_path="mlp_trt_int8.trt",
    calib_data=calib_data
)
