1. nn.Module — The Foundation Must be instant

Basic pattern

import torch
import torch.nn as nn
import torch.nn.functional as F

class MLP(nn.Module):
    def __init__(self, d_in, d_hidden, d_out, dropout=0.1):
        super().__init__()
        self.fc1 = nn.Linear(d_in, d_hidden)
        self.fc2 = nn.Linear(d_hidden, d_out)
        self.dropout = nn.Dropout(dropout)
        self.norm = nn.LayerNorm(d_hidden)

    def forward(self, x):
        # x: (B, T, d_in)
        x = self.fc1(x)           # (B, T, d_hidden)
        x = F.gelu(x)
        x = self.norm(x)
        x = self.dropout(x)
        x = self.fc2(x)           # (B, T, d_out)
        return x
Always: super().__init__(), define layers in __init__, compute in forward. Shapes in comments. Never call .backward() inside a Module.

2. Multi-Head Self-Attention High probability

class MultiHeadSelfAttention(nn.Module):
    def __init__(self, d_model, n_heads, dropout=0.1):
        super().__init__()
        assert d_model % n_heads == 0
        self.d_model = d_model
        self.n_heads = n_heads
        self.d_k = d_model // n_heads

        # Single projection for Q, K, V (more efficient)
        self.qkv = nn.Linear(d_model, 3 * d_model)
        self.out = nn.Linear(d_model, d_model)
        self.dropout = nn.Dropout(dropout)
        self.scale = self.d_k ** -0.5

    def forward(self, x, mask=None):
        # x: (B, T, D)
        B, T, D = x.shape

        # Project and split into heads
        qkv = self.qkv(x)                           # (B, T, 3D)
        qkv = qkv.reshape(B, T, 3, self.n_heads, self.d_k)
        qkv = qkv.permute(2, 0, 3, 1, 4)          # (3, B, H, T, d_k)
        q, k, v = qkv[0], qkv[1], qkv[2]             # each (B, H, T, d_k)

        # Attention scores
        attn = (q @ k.transpose(-2, -1)) * self.scale  # (B, H, T, T)

        if mask is not None:
            attn = attn.masked_fill(mask == 0, float('-inf'))

        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)

        # Weighted sum
        out = attn @ v                                # (B, H, T, d_k)
        out = out.transpose(1, 2).reshape(B, T, D)    # (B, T, D)
        out = self.out(out)
        return out
Common mistakes: (1) Forgetting * self.scale — scores explode, softmax saturates. (2) Wrong permute order — always trace shapes in comments. (3) Forgetting to reshape back after attention.

3. Cross-Attention Nuance-specific

Audio conditioning the video generator uses cross-attention. Q comes from video features, K and V come from audio features.

class CrossAttention(nn.Module):
    """Q from target modality, K/V from conditioning modality."""
    def __init__(self, d_model, d_cond, n_heads, dropout=0.1):
        super().__init__()
        assert d_model % n_heads == 0
        self.n_heads = n_heads
        self.d_k = d_model // n_heads

        self.q_proj = nn.Linear(d_model, d_model)
        self.k_proj = nn.Linear(d_cond, d_model)    # K from conditioning
        self.v_proj = nn.Linear(d_cond, d_model)    # V from conditioning
        self.out = nn.Linear(d_model, d_model)
        self.scale = self.d_k ** -0.5

    def forward(self, x, cond):
        # x: (B, T_video, D)  — video features (query)
        # cond: (B, T_audio, D_cond)  — audio features (key/value)
        B, T_v, D = x.shape
        T_a = cond.shape[1]

        q = self.q_proj(x).reshape(B, T_v, self.n_heads, self.d_k).transpose(1, 2)
        k = self.k_proj(cond).reshape(B, T_a, self.n_heads, self.d_k).transpose(1, 2)
        v = self.v_proj(cond).reshape(B, T_a, self.n_heads, self.d_k).transpose(1, 2)
        # q: (B, H, T_v, d_k), k/v: (B, H, T_a, d_k)

        attn = (q @ k.transpose(-2, -1)) * self.scale  # (B, H, T_v, T_a)
        attn = F.softmax(attn, dim=-1)

        out = attn @ v                                  # (B, H, T_v, d_k)
        out = out.transpose(1, 2).reshape(B, T_v, D)
        return self.out(out)
Self-attention vs cross-attention — the only difference is where K,V come from. Self: K,V from same input as Q. Cross: K,V from a different modality (audio conditioning video).

4. DiT Block (Diffusion Transformer) Nuance architecture

Their model is a dual-tower DiT with 32-64 flow matching steps. A DiT block = self-attention + cross-attention + FFN, conditioned on timestep via adaptive layer norm (adaLN).

class AdaLayerNorm(nn.Module):
    """LayerNorm whose scale/shift are predicted from timestep embedding."""
    def __init__(self, d_model, d_cond):
        super().__init__()
        self.norm = nn.LayerNorm(d_model, elementwise_affine=False)
        self.proj = nn.Linear(d_cond, 2 * d_model)  # predict scale and shift

    def forward(self, x, t_emb):
        # x: (B, T, D), t_emb: (B, D_cond)
        scale_shift = self.proj(t_emb).unsqueeze(1)  # (B, 1, 2D)
        scale, shift = scale_shift.chunk(2, dim=-1)  # each (B, 1, D)
        return self.norm(x) * (1 + scale) + shift


class DiTBlock(nn.Module):
    """One block of a Diffusion Transformer."""
    def __init__(self, d_model, n_heads, d_cond, d_audio, dropout=0.1):
        super().__init__()
        # AdaLN for timestep conditioning
        self.norm1 = AdaLayerNorm(d_model, d_cond)
        self.norm2 = AdaLayerNorm(d_model, d_cond)
        self.norm3 = AdaLayerNorm(d_model, d_cond)

        # Self-attention over spatial patches
        self.self_attn = MultiHeadSelfAttention(d_model, n_heads, dropout)

        # Cross-attention: video attends to audio
        self.cross_attn = CrossAttention(d_model, d_audio, n_heads, dropout)

        # FFN
        self.ffn = nn.Sequential(
            nn.Linear(d_model, 4 * d_model),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(4 * d_model, d_model),
            nn.Dropout(dropout),
        )

    def forward(self, x, t_emb, audio_cond):
        # x: (B, N_patches, D) — noisy video latent patches
        # t_emb: (B, D_cond) — timestep embedding
        # audio_cond: (B, T_audio, D_audio) — audio features

        x = x + self.self_attn(self.norm1(x, t_emb))
        x = x + self.cross_attn(self.norm2(x, t_emb), audio_cond)
        x = x + self.ffn(self.norm3(x, t_emb))
        return x
DiT key insight: x is a sequence of spatial patches (like ViT), NOT sequential frames. All patches processed in parallel per denoising step. The timestep t enters via adaLN, not concatenation.

5. Training Loop Must be instant

def train_one_epoch(model, dataloader, optimizer, device):
    model.train()
    total_loss = 0.0

    for batch in dataloader:
        x, y = batch
        x, y = x.to(device), y.to(device)

        # Forward
        logits = model(x)
        loss = F.cross_entropy(logits, y)

        # Backward
        optimizer.zero_grad()       # MUST come before backward
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
        optimizer.step()

        total_loss += loss.item()

    return total_loss / len(dataloader)
Order matters: zero_grad() → backward() → clip_grad → step(). Getting this wrong is an instant red flag.

With mixed precision (AMP)

from torch.amp import autocast, GradScaler

scaler = GradScaler()

for batch in dataloader:
    x, y = batch[0].to(device), batch[1].to(device)
    optimizer.zero_grad()

    with autocast(device_type='cuda', dtype=torch.float16):
        logits = model(x)
        loss = F.cross_entropy(logits, y)

    scaler.scale(loss).backward()
    scaler.unscale_(optimizer)
    torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
    scaler.step(optimizer)
    scaler.update()
Why AMP matters for Nuance: Their DiT is 1-5B params. FP16 training with GradScaler halves memory, doubles throughput. Mention this if they ask about training efficiency.

6. Flow Matching Training Step Their exact paradigm

Nuance uses 32-64 flow matching steps. Flow matching learns a velocity field v(x_t, t) that transports noise to data along a straight path.

def flow_matching_loss(model, x_0, audio_cond):
    """One training step for flow matching (conditional flow matching / rectified flow)."""
    B = x_0.shape[0]
    device = x_0.device

    # 1. Sample random timestep t ~ U(0, 1)
    t = torch.rand(B, device=device)                    # (B,)

    # 2. Sample noise
    noise = torch.randn_like(x_0)                        # (B, ...)

    # 3. Interpolate: x_t = (1-t)*noise + t*x_0  (straight path)
    t_expand = t.view(B, *([1] * (x_0.dim() - 1)))    # broadcast
    x_t = (1 - t_expand) * noise + t_expand * x_0

    # 4. Target velocity: v_target = x_0 - noise  (direction of the straight path)
    v_target = x_0 - noise

    # 5. Predict velocity
    v_pred = model(x_t, t, audio_cond)                   # model predicts velocity

    # 6. Loss: MSE between predicted and target velocity
    loss = F.mse_loss(v_pred, v_target)
    return loss

Flow matching inference (ODE solver)

@torch.no_grad()
def flow_matching_sample(model, shape, audio_cond, n_steps=32):
    """Generate samples by integrating the learned velocity field."""
    device = audio_cond.device

    # Start from pure noise (t=0)
    x = torch.randn(shape, device=device)

    # Euler integration from t=0 to t=1
    dt = 1.0 / n_steps
    for i in range(n_steps):
        t = torch.full((shape[0],), i * dt, device=device)
        v = model(x, t, audio_cond)
        x = x + v * dt

    return x  # clean sample at t=1
Flow matching vs diffusion in code: Flow matching: linear interpolation, velocity target, Euler ODE. Diffusion: noise schedule, noise prediction, iterative denoising. Flow matching is simpler code, fewer steps, straighter paths.

7. Diffusion Training Step Know both paradigms

def diffusion_loss(model, x_0, audio_cond, n_timesteps=1000):
    """DDPM-style diffusion training step."""
    B = x_0.shape[0]
    device = x_0.device

    # 1. Sample random integer timestep
    t = torch.randint(0, n_timesteps, (B,), device=device)

    # 2. Noise schedule (simplified linear)
    beta = torch.linspace(1e-4, 0.02, n_timesteps, device=device)
    alpha = 1.0 - beta
    alpha_bar = torch.cumprod(alpha, dim=0)            # cumulative product
    alpha_bar_t = alpha_bar[t].view(B, *([1] * (x_0.dim() - 1)))

    # 3. Add noise: x_t = sqrt(alpha_bar_t) * x_0 + sqrt(1 - alpha_bar_t) * noise
    noise = torch.randn_like(x_0)
    x_t = alpha_bar_t.sqrt() * x_0 + (1 - alpha_bar_t).sqrt() * noise

    # 4. Predict the noise
    noise_pred = model(x_t, t, audio_cond)

    # 5. Loss: MSE between predicted and actual noise
    loss = F.mse_loss(noise_pred, noise)
    return loss

8. K-Means in PyTorch Proven interview question

You got time-killed on this at Netflix after 50 min drain. The algorithm is simple — drill it until it's muscle memory. No sklearn, no sorting. Just assign + update.
def kmeans(data, k, max_iters=100, tol=1e-4):
    """
    K-means clustering in pure PyTorch.
    data: (N, D) tensor
    k: number of clusters
    Returns: centroids (K, D), assignments (N,)
    """
    N, D = data.shape

    # 1. Initialize: random selection from data
    indices = torch.randperm(N)[:k]
    centroids = data[indices].clone()                # (K, D)

    for _ in range(max_iters):
        # 2. ASSIGN: each point to nearest centroid
        #    dist[i,j] = ||data[i] - centroids[j]||^2
        dists = torch.cdist(data, centroids)         # (N, K)
        assignments = dists.argmin(dim=1)            # (N,) — NOT sort, argmin

        # 3. UPDATE: new centroids = mean of assigned points
        new_centroids = torch.zeros_like(centroids)
        for j in range(k):
            mask = (assignments == j)
            if mask.sum() > 0:
                new_centroids[j] = data[mask].mean(dim=0)
            else:
                new_centroids[j] = data[torch.randint(N, (1,))]  # reinit empty

        # 4. CONVERGE: check if centroids moved
        shift = (new_centroids - centroids).norm()
        centroids = new_centroids
        if shift < tol:
            break

    return centroids, assignments

Vectorized update (no loop over k)

# Replace the for-loop update with scatter:
def update_centroids_vectorized(data, assignments, k):
    # One-hot: (N, K)
    one_hot = F.one_hot(assignments, k).float()      # (N, K)
    counts = one_hot.sum(dim=0).clamp(min=1)          # (K,) avoid /0
    centroids = (one_hot.T @ data) / counts.unsqueeze(1)  # (K, D)
    return centroids
Complexity: O(iters × N × K × D). The key operations: torch.cdist for distances, argmin for assignment (NOT sort), mean for update. If asked about initialization: K-means++ picks centroids far apart, reduces bad local optima.

10. Quantization in PyTorch Your specialty

Post-training quantization (PTQ) — basic

def quantize_tensor(x, n_bits=8):
    """Symmetric uniform quantization."""
    qmin = -(2 ** (n_bits - 1))
    qmax = 2 ** (n_bits - 1) - 1

    # Scale: maps float range to int range
    scale = x.abs().max() / qmax

    # Quantize: float -> int
    x_q = (x / scale).round().clamp(qmin, qmax)

    # Dequantize: int -> float (for computation)
    x_deq = x_q * scale
    return x_deq, scale


def quantize_per_channel(weight, n_bits=8):
    """Per-channel quantization (better accuracy than per-tensor)."""
    # weight: (out_features, in_features)
    qmax = 2 ** (n_bits - 1) - 1
    scale = weight.abs().amax(dim=1, keepdim=True) / qmax  # (out, 1)
    w_q = (weight / scale).round().clamp(-qmax - 1, qmax)
    return w_q * scale, scale

Quantization-aware training (QAT) — your core

class StraightThroughQuantize(torch.autograd.Function):
    """STE: forward quantizes, backward passes gradient through."""
    @staticmethod
    def forward(ctx, x, scale, qmin, qmax):
        x_q = (x / scale).round().clamp(qmin, qmax)
        return x_q * scale

    @staticmethod
    def backward(ctx, grad_output):
        # Straight-through: pass gradient unchanged
        return grad_output, None, None, None


class QuantizedLinear(nn.Module):
    """Linear layer with fake quantization during training."""
    def __init__(self, in_f, out_f, n_bits=8):
        super().__init__()
        self.linear = nn.Linear(in_f, out_f)
        self.n_bits = n_bits
        self.qmax = 2 ** (n_bits - 1) - 1

    def forward(self, x):
        # Quantize weights (fake quant with STE)
        scale = self.linear.weight.abs().max() / self.qmax
        w_q = StraightThroughQuantize.apply(
            self.linear.weight, scale, -self.qmax - 1, self.qmax
        )
        return F.linear(x, w_q, self.linear.bias)
STE (Straight-Through Estimator) is the key trick for QAT. round() has zero gradient everywhere. STE pretends the gradient of round() is 1. This is the foundation of all quantization-aware training, including your S8BQAT and GQ.

11. VAE Reparameterization Foundation

class VAE(nn.Module):
    def __init__(self, d_in, d_latent):
        super().__init__()
        self.encoder = nn.Linear(d_in, 2 * d_latent)  # outputs mu and log_var
        self.decoder = nn.Linear(d_latent, d_in)

    def encode(self, x):
        h = self.encoder(x)
        mu, log_var = h.chunk(2, dim=-1)
        return mu, log_var

    def reparameterize(self, mu, log_var):
        # z = mu + sigma * epsilon, epsilon ~ N(0,1)
        std = (log_var * 0.5).exp()
        eps = torch.randn_like(std)
        return mu + std * eps   # gradients flow through mu and std

    def forward(self, x):
        mu, log_var = self.encode(x)
        z = self.reparameterize(mu, log_var)
        recon = self.decoder(z)
        return recon, mu, log_var


def vae_loss(recon, x, mu, log_var):
    # Reconstruction loss
    recon_loss = F.mse_loss(recon, x, reduction='sum')
    # KL divergence: D_KL(q(z|x) || p(z)) where p(z) = N(0,1)
    kl_loss = -0.5 * (1 + log_var - mu.pow(2) - log_var.exp()).sum()
    return recon_loss + kl_loss

12. Sinusoidal Timestep Embedding Used in DiT/diffusion

def sinusoidal_embedding(t, d_model):
    """
    Sinusoidal positional/timestep embedding.
    t: (B,) integer or float timesteps
    Returns: (B, d_model)
    """
    half = d_model // 2
    freqs = torch.exp(
        -torch.arange(half, device=t.device).float() * (torch.log(torch.tensor(10000.0)) / half)
    )  # (d_model/2,)

    args = t.unsqueeze(1).float() * freqs.unsqueeze(0)   # (B, d_model/2)
    emb = torch.cat([args.sin(), args.cos()], dim=-1)    # (B, d_model)
    return emb
Same formula as transformer positional encoding, but applied to timestep t instead of position. This is how the model knows which denoising step it's at.

13. GQA (Grouped Query Attention) Compression topic

class GroupedQueryAttention(nn.Module):
    """GQA: G groups of heads share K,V. Reduces KV cache by n_heads/n_kv_heads."""
    def __init__(self, d_model, n_heads, n_kv_heads):
        super().__init__()
        assert n_heads % n_kv_heads == 0
        self.n_heads = n_heads
        self.n_kv_heads = n_kv_heads
        self.group_size = n_heads // n_kv_heads
        self.d_k = d_model // n_heads

        self.q_proj = nn.Linear(d_model, d_model)                      # full heads for Q
        self.k_proj = nn.Linear(d_model, self.n_kv_heads * self.d_k)   # fewer heads for K
        self.v_proj = nn.Linear(d_model, self.n_kv_heads * self.d_k)   # fewer heads for V
        self.out = nn.Linear(d_model, d_model)
        self.scale = self.d_k ** -0.5

    def forward(self, x):
        B, T, D = x.shape

        q = self.q_proj(x).reshape(B, T, self.n_heads, self.d_k).transpose(1, 2)
        k = self.k_proj(x).reshape(B, T, self.n_kv_heads, self.d_k).transpose(1, 2)
        v = self.v_proj(x).reshape(B, T, self.n_kv_heads, self.d_k).transpose(1, 2)
        # q: (B, n_heads, T, d_k), k/v: (B, n_kv_heads, T, d_k)

        # Expand K,V to match Q heads via repeat
        k = k.repeat_interleave(self.group_size, dim=1)  # (B, n_heads, T, d_k)
        v = v.repeat_interleave(self.group_size, dim=1)

        attn = (q @ k.transpose(-2, -1)) * self.scale
        attn = F.softmax(attn, dim=-1)
        out = (attn @ v).transpose(1, 2).reshape(B, T, D)
        return self.out(out)
GQA ratio: n_heads=32, n_kv_heads=8 → 4x KV cache reduction. Llama 2 70B uses this. n_kv_heads=1 is MQA (Multi-Query Attention). The repeat_interleave is the key line — it broadcasts KV across query head groups.

14. Normalization Variants Quick reference

# LayerNorm: normalize over feature dim (standard in transformers)
ln = nn.LayerNorm(d_model)        # normalizes last dim

# RMSNorm: LayerNorm without mean centering (Llama, modern LLMs)
class RMSNorm(nn.Module):
    def __init__(self, d, eps=1e-6):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(d))
        self.eps = eps

    def forward(self, x):
        rms = (x.pow(2).mean(dim=-1, keepdim=True) + self.eps).rsqrt()
        return x * rms * self.weight

# BatchNorm: normalize over batch dim (CNNs, NOT transformers)
bn = nn.BatchNorm1d(d_model)      # breaks for AR inference (batch=1)
